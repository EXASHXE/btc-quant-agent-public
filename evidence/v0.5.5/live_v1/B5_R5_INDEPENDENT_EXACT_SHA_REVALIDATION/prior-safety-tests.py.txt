@pytest.mark.parametrize('caller_delta', [-999999, 0, 999999])
@pytest.mark.parametrize('late', [0, 1, 1000])
def test_actual_clock_overrules_caller_time(tmp_path, caller_delta, late):
    backend, _, _, _, client, receipt, clock = placement(tmp_path)
    clock['now'] = receipt.expires_at_ms + late
    with pytest.raises(ExecutionBlocked):
        backend.submit_authorized(receipt.authorization_id, NOW+caller_delta)
    for name in ('change_margin_type', 'change_leverage', 'place_order', 'place_protective_order'):
        getattr(client, name).assert_not_called()

@pytest.mark.parametrize('field,value', [('daily_loss_usdt',24.9999), ('drawdown_pct',.0499999), ('available_balance_usdt',.000001), ('reconciled',False), ('stream_connected',False)])
def test_second_forced_account_read_blocks_entry(tmp_path, field, value):
    backend, _, _, account, client, receipt, _ = placement(tmp_path)
    bad = fresh_account(account, **{field:value})
    backend.account_provider.side_effect = [account, bad]
    with pytest.raises(ExecutionBlocked):
        backend.submit_authorized(receipt.authorization_id, NOW-100000)
    assert backend.account_provider.call_count == 2
    client.place_order.assert_not_called()

@pytest.mark.parametrize('field', ['intent_id','idempotency_key','client_order_id'])
def test_self_rehash_cannot_replace_canonical_ids(tmp_path, field):
    _, intent, _, _ = setup_validator_and_intent(tmp_path)
    raw = intent.model_dump(mode='json')
    raw[field] = 'independent-foreign-id'
    raw['intent_hash'] = content_hash({k:v for k,v in raw.items() if k != 'intent_hash'})
    with pytest.raises(ValueError):
        TradeIntentV1.model_validate(raw)
    digest = hashlib.sha256(f'{intent.case_hash}:{intent.proposal_hash}:{intent.approval_event_id}'.encode()).hexdigest()
    assert (intent.intent_id,intent.idempotency_key,intent.client_order_id)==(f'intent_{digest[:24]}',f'idem_{digest[:24]}',f'cuid_v1_{digest[:16]}')

@pytest.mark.parametrize('component', ['mark','book','kline'])
@pytest.mark.parametrize('timestamp', ['source','receipt'])
@pytest.mark.parametrize('delta', [-16000, 1])
def test_each_market_component_has_independent_freshness(tmp_path, component, timestamp, delta):
    validator, intent, snap, _ = setup_validator_and_intent(tmp_path)
    market = sample_market_obs()
    field = f'{component}_{timestamp}_timestamp_ms'
    raw = market.model_dump(exclude={'observation_hash'})
    raw[field] = NOW+delta
    modified = type(market).build(**raw)
    result = validator.validate(intent, modified, snap, NOW)
    assert not result.is_valid
    assert f'MARKET_{component.upper()}' in result.reason

@pytest.mark.parametrize('filled', [False, True])
def test_recovery_precedes_new_risk_gates_after_restart(tmp_path, filled):
    service, backend, intent, client = recovered_service(tmp_path, filled=filled)
    backend.kill_switch.allows_new_risk = lambda: False
    backend.validator.tactical_validity_provider = lambda *_: False
    for _ in range(2):
        result = asyncio.run(service.execute_approved_intent(intent.intent_id, intent.expires_at_ms+9999))
        assert result.filled_qty == (intent.quantity/2 if filled else 0)
    client.place_order.assert_not_called()
    client.change_margin_type.assert_not_called()
    if filled:
        client.place_protective_order.assert_called_once()
        assert client.place_protective_order.call_args.kwargs['quantity'] == intent.quantity/2

def test_transport_uncertainty_never_reposts_on_restart(tmp_path):
    backend, _, intent, _, client, receipt, _ = placement(tmp_path)
    client.place_order.side_effect = TimeoutError('local simulated lost response')
    client.query_order_by_client_id.side_effect = BinanceExecutionError('local uncertain exact query')
    with pytest.raises(ExecutionBlocked):
        backend.submit_authorized(receipt.authorization_id, NOW)
    for _ in range(2):
        with pytest.raises(ExecutionBlocked):
            backend.submit_authorized(receipt.authorization_id, NOW+1)
    client.place_order.assert_called_once()
    assert client.query_order_by_client_id.call_count == 3
    assert backend.validator.intent_store.has_existing_side_effect(intent.intent_id)

@pytest.mark.parametrize('environment,namespace', [('DRY_RUN','NONE'),('TESTNET','BINANCE_TESTNET')])
@pytest.mark.parametrize('account', ['validator-a','validator-b'])
def test_flat_reopen_lifecycle_survives_restart(tmp_path, environment, namespace, account):
    path=tmp_path/'lifecycle.db'
    changes=dict(account_id=account,environment=environment,credential_namespace=namespace)
    first=PositionSupervisor(path)
    first.evaluate(observation(**changes,quantity=-.25,previous_quantity=0),NOW)
    first.evaluate(observation(**changes,quantity=0,observed_at_ms=NOW+1),NOW+1)
    restarted=PositionSupervisor(path)
    assert restarted.current_quantity(account,'BTCUSDT',environment,namespace)==0
    events=restarted.evaluate(observation(**changes,quantity=.4,observed_at_ms=NOW+2),NOW+2)
    assert len([e for e in events if e.trigger=='POSITION_OPENED'])==1
    assert restarted.current_quantity(account,'BTCUSDT',environment,namespace)==.4

def test_cooldown_cannot_cross_account_or_environment(tmp_path):
    supervisor=PositionSupervisor(tmp_path/'cooldown.db')
    for i,(account,env,ns) in enumerate([('A','DRY_RUN','NONE'),('B','DRY_RUN','NONE'),('A','TESTNET','BINANCE_TESTNET')]):
        events=supervisor.evaluate(observation(account_id=account,environment=env,credential_namespace=ns,mark_price=91,observed_at_ms=NOW+i),NOW+i)
        assert 'STOP_NEAR' in [e.trigger for e in events]
    events=PositionSupervisor(tmp_path/'cooldown.db').evaluate(observation(account_id='A',mark_price=91,observed_at_ms=NOW+3),NOW+3)
    assert 'STOP_NEAR' not in [e.trigger for e in events]

def test_sqlite_contention_and_rollback_preserve_durable_state(tmp_path):
    path=tmp_path/'durable.db'
    with connection(path) as db:
        assert db.execute('PRAGMA synchronous').fetchone()[0]==2
        assert db.execute('PRAGMA journal_mode').fetchone()[0]=='wal'
        db.execute('CREATE TABLE independent_events(id INTEGER PRIMARY KEY)')
    def write(i):
        with connection(path) as db:
            db.execute('BEGIN IMMEDIATE')
            db.execute('INSERT INTO independent_events VALUES(?)',(i,))
    with ThreadPoolExecutor(max_workers=4) as pool:
        list(pool.map(write,range(24)))
    with pytest.raises(RuntimeError):
        with connection(path) as db:
            db.execute('INSERT INTO independent_events VALUES(999)')
            raise RuntimeError('local abort')
    with connection(path) as held:
        held.execute('BEGIN IMMEDIATE')
        with pytest.raises(sqlite3.OperationalError):
            with connection(path,busy_timeout_ms=20) as contender:
                contender.execute('INSERT INTO independent_events VALUES(998)')
    with connection(path) as db:
        assert db.execute('SELECT COUNT(*) FROM independent_events').fetchone()[0]==24

@pytest.mark.parametrize('b3_enabled',[False,True])
def test_b4_host_routes_have_read_only_authority(tmp_path,monkeypatch,b3_enabled):
    monkeypatch.setattr(api,'load_config',lambda:AppConfig(storage=StorageConfig(sqlite_path=str(tmp_path/'api.db'))))
    monkeypatch.setenv('BTC_QUANT_LIVE_V1_ENABLED','true' if b3_enabled else 'false')
    monkeypatch.setenv('BTC_QUANT_LIVE_V1_RUNTIME_ENABLED','true')
    service=TacticalLiveService(LiveStore(tmp_path/'live.db'),None,RiskCompilerV1(RiskPolicyV1()))
    runtime=SimpleNamespace(tactical_service=service,start=AsyncMock(),stop=AsyncMock(),status=Mock(return_value={}),execute_intent=AsyncMock())
    app=api.create_app(live_runtime=runtime)
    for route in app.routes:
        assert set(getattr(route,'methods',[])) <= {'GET','HEAD','OPTIONS'}
        assert getattr(route,'path','') not in {'/live-v1/scan','/live-v1/reviews'}
    runtime.execute_intent.assert_not_awaited()

@pytest.mark.parametrize('field,value',[('quantity',999),('side','SELL'),('stop_loss',1),('leverage',125),('execution_mode','LIVE'),('intent_id','foreign')])
def test_feishu_payload_cannot_supply_executable_authority(tmp_path,field,value):
    store,case,_,proposal=prepared(tmp_path)
    app=create_callback_app(store,'verify',approver_open_ids=('allowed',),app_id='app',clock_ms=lambda:3000)
    async def request():
        async with httpx.AsyncClient(transport=httpx.ASGITransport(app=app),base_url='http://fake') as client:
            response=await client.post('/callback',json=callback_body(case,proposal,value_extra={field:value}))
            assert response.status_code==400
    asyncio.run(request())
    assert store.state(case.case_id)=='WAITING_APPROVAL'
